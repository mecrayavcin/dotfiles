/dev/nvme0n1p3 Linux filesystem olacak

# Root sifreleme
cryptsetup luksFormat /dev/nvme0n1p3
(Sifreni belirle, onay YES buyuk harflerle olmali)

# Root Sifreli bolum acma
cryptsetup open --type luks /dev/nvme0n1p3 lvm
(Sife gir ac)

# Olusan lvm gorme
ls -l /dev/mapper
(Lvm gozukmeli)

# lvm duzenleme disk boyutu belirleme
pvcreate /dev/mapper/lvm
(Bu cihazı LVM için “fiziksel hacim” (Physical Volume) olarak işaretler)

vgcreate volgroup0 /dev/mapper/lvm
(Havuz olusturuldu)

lvcreate -L 80GB volgroup0 -n lv_root
(80 GB / kok dizin olusturuldu)

lvcreate -l 100%FREE volgroup0 -n lv_home
(kalan GB /home dizin olusturuldu)

modprobe dm_mod
(Device Mapper çekirdek modülünü yükler)

vgscan
(Sistemdeki volume group’ları arar, birsey yapmana gerek yok)

vgchange -ay
(Gruplari aktif eder, birsey yapmana gerek yok)

# Formatlama ve MOUNT

(fdisk -l ile parca ve isimleri gorebilirsin)

mkfs.ext4 /dev/volgroup0/lv_home

mkswap /dev/nvme0n1p2

mount /dev/volgroup0/lv_root /mnt

mkdir /mnt/home
mount /dev/volgroup0/lv_home /mnt/home
